// 画像の余白と歪みを正確に補正した左右対称・位置ジャストフィット座標算出関数 function getTileCoordinates(id) { // 外枠黒線の内側（ボード本体）の各端位置（%） const leftEdge = 7.1; // 左端マスの中心 const rightEdge = 92.9; // 右端マスの中心 const topEdge = 7.1; // 上端マスの中心 const bottomEdge = 92.9;// 下端マスの中心 const stepX = (rightEdge - leftEdge) / 10; // 8.58% (横方向の1マス幅) const stepY = (bottomEdge - topEdge) / 10; // 8.58% (縦方向の1マス幅) // 四隅コーナーマス if (id === 0) return { left: rightEdge, top: bottomEdge }; // GO (右下) if (id === 10) return { left: leftEdge, top: bottomEdge }; // 刑務所 (左下) if (id === 20) return { left: leftEdge, top: topEdge }; // 無料駐車場 (左上) if (id === 30) return { left: rightEdge, top: topEdge }; // Go to Jail (右上) // 下辺 (1〜9): 右から左へ if (id >= 1 && id <= 9) { return { left: rightEdge - (id * stepX), top: bottomEdge }; } // 左辺 (11〜19): 下から上へ if (id >= 11 && id <= 19) { return { left: leftEdge, top: bottomEdge - ((id - 10) * stepY) }; } // 上辺 (21〜29): 左から右へ if (id >= 21 && id <= 29) { return { left: leftEdge + ((id - 20) * stepX), top: topEdge }; } // 右辺 (31〜39): 上から下へ if (id >= 31 && id <= 39) { return { left: rightEdge, top: topEdge + ((id - 30) * stepY) }; } }
